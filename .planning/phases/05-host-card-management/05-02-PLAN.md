struct ClipPreviewSheet: View {
    let clip: Clip
    let onDelete: () async -> Void

    @State private var signedURL: URL?
    @State private var isLoading = true
    @State private var loadError: String?
    @State private var showDeleteConfirmation = false
    @State private var player: AVPlayer?
    @Environment(\.dismiss) private var dismiss

    private let storageService = StorageService()

    var body: some View {
        NavigationStack {
            VStack(spacing: 0) {
                // Video player area
                ZStack {
                    if let url = signedURL, let player {
                        LoopingVideoPlayer(player: player)  // Reuse from VideoPreviewView
                            .aspectRatio(9/16, contentMode: .fit)
                    } else if isLoading {
                        Rectangle()
                            .fill(Color.black)
                            .aspectRatio(9/16, contentMode: .fit)
                            .overlay { ProgressView().tint(.white) }
                    } else if let error = loadError {
                        Rectangle()
                            .fill(Color.black)
                            .aspectRatio(9/16, contentMode: .fit)
                            .overlay {
                                VStack(spacing: 8) {
                                    Image(systemName: "exclamationmark.triangle")
                                        .font(.largeTitle)
                                        .foregroundColor(.red)
                                    Text("Failed to load video")
                                        .foregroundColor(.white)
                                }
                            }
                    }
                }
                .clipShape(RoundedRectangle(cornerRadius: 16))
                .padding(.horizontal, 24)

                Spacer()

                // Delete button
                TOYButton("Delete Clip", style: .destructive, size: .large) {
                    showDeleteConfirmation = true
                }
                .padding(.horizontal, 24)
                .padding(.bottom, 32)
            }
            .background(Color.toyBackground)
            .navigationTitle("Preview Clip")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .topBarTrailing) {
                    Button("Done") { dismiss() }
                }
            }
            .task { await loadSignedURL() }
            .onDisappear {
                player?.pause()
                player = nil
            }
            .confirmationDialog("Delete Clip", isPresented: $showDeleteConfirmation, titleVisibility: .visible) {
                Button("Delete", role: .destructive) {
                    Task {
                        await onDelete()
                        dismiss()
                    }
                }
            } message: {
                Text("This clip will be permanently removed from the card.")
            }
        }
    }

    private func loadSignedURL() async {
        do {
            let url = try await storageService.createSignedURL(path: clip.videoUrl)
            signedURL = url
            setupPlayer(url: url)
        } catch {
            loadError = error.localizedDescription
            #if DEBUG
            print("❌ Failed to get signed URL: \(error)")
            #endif
        }
        isLoading = false
    }

    private func setupPlayer(url: URL) {
        let newPlayer = AVPlayer(url: url)
        newPlayer.play()

        NotificationCenter.default.addObserver(
            forName: .AVPlayerItemDidPlayToEndTime,
            object: newPlayer.currentItem,
            queue: .main
        ) { _ in
            newPlayer.seek(to: .zero)
            newPlayer.play()
        }
        player = newPlayer
    }
}

struct CardDetailView: View {
    let card: Card

    @State private var viewModel = CardDetailViewModel()
    @State private var selectedClip: Clip?

    var body: some View {
        List {
            // Card info section
            Section {
                VStack(alignment: .leading, spacing: 4) {
                    TOYLabel(card.recipientName, style: .headline)
                    TOYLabel("For: \(card.recipientName)", style: .subheadline, color: .toyTextSecondary)
                    if let occasion = card.occasion {
                        TOYLabel(occasion, style: .caption, color: .toyTextSecondary)
                    }
                }
                .listRowBackground(Color.toySurface)
            }

            // Participants section
            Section("Participants (\(viewModel.participants.count))") {
                if viewModel.participants.isEmpty && !viewModel.isLoading {
                    Text("No participants invited yet")
                        .foregroundColor(.toyTextSecondary)
                } else {
                    ForEach(viewModel.participants) { participant in
                        ParticipantRow(participant: participant)
                    }
                }
            }

            // Clips section
            Section("Submitted Clips (\(viewModel.clips.count))") {
                if viewModel.clips.isEmpty && !viewModel.isLoading {
                    Text("No clips submitted yet")
                        .foregroundColor(.toyTextSecondary)
                } else {
                    ForEach(viewModel.clips) { clip in
                        ClipRow(clip: clip) {
                            selectedClip = clip
                        }
                    }
                }
            }
        }
        .listStyle(.insetGrouped)
        .navigationTitle(card.title)
        .refreshable {
            await viewModel.loadData(for: card.id)
        }
        .task {
            await viewModel.loadData(for: card.id)
        }
        .overlay {
            if viewModel.isLoading && viewModel.participants.isEmpty {
                ProgressView()
            }
        }
        .sheet(item: $selectedClip) { clip in
            ClipPreviewSheet(clip: clip) {
                await viewModel.deleteClip(clip)
            }
        }
        .alert("Error", isPresented: .constant(viewModel.errorMessage != nil)) {
            Button("OK") { viewModel.errorMessage = nil }
        } message: {
            Text(viewModel.errorMessage ?? "")
        }
    }
}

// Helper view for clip rows
private struct ClipRow: View {
    let clip: Clip

    var body: some View {

                        TOYLabel("\(duration)s", style: .caption, color: .toyTextSecondary)

                    TOYLabel(clip.status.capitalized, style: .caption, color: .toyTextSecondary)
                }

                Spacer()

                Image(systemName: "chevron.right")
                    .foregroundColor(.toyTextSecondary)
            }
        }
        .buttonStyle(.plain)
    }
}
