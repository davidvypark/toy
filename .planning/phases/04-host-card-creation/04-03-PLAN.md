**Modify performUpload():**
After successful upload (when setting uploadState = .success), add clip record creation:

```swift
// After: uploadedPath = path
// After: uploadState = .success(storagePath: path)

// Create clip record if we have card context
if let cardId = cardId, let participantId = participantId {
    do {
        let orderPosition = isHostClip ? 0 : nil  // Host clip = 0, participants assigned later
        let clip = try await cardService.createClip(
            cardId: cardId,
            participantId: participantId,
            videoUrl: path,
            durationSeconds: nil,  // Could calculate from video, but not critical for MVP
            orderPosition: orderPosition ?? 1,  // Default to 1 for non-host
            status: "uploaded"
        )
        createdClip = clip

        // If host clip, update card status to 'collecting'
        if isHostClip {
            try await cardService.updateCardStatus(cardId: cardId, status: "collecting")
        }
    } catch {
        print("Failed to create clip record: \(error)")
        // Don't fail the upload - video is already stored
    }
}
```

**Important:** The clip record creation is best-effort. If it fails, the video is still uploaded. This prevents frustrating UX where upload succeeds but clip record fails.
