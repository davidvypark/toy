    import Foundation
    import RevenueCat

    @MainActor
    @Observable
    public class UpgradeViewModel {
        public enum State: Equatable {
            case idle
            case loading
            case ready(Package)
            case purchasing
            case success
            case error(String)
        }

        public private(set) var state: State = .idle

        private let cardId: UUID
        private let purchaseService = PurchaseService.shared

        public init(cardId: UUID) {
            self.cardId = cardId
        }

        /// Load available upgrade package from RevenueCat
        public func loadOffering() async {
            state = .loading
            do {
                let offerings = try await purchaseService.fetchOfferings()
                // Look for "card_upgrade" offering or default offering
                guard let offering = offerings.current,
                      let package = offering.availablePackages.first else {
                    state = .error("No upgrade package available")
                    return
                }
                state = .ready(package)
            } catch {
                state = .error(error.localizedDescription)
            }
        }

        /// Purchase the upgrade package
        public func purchase() async {
            guard case .ready(let package) = state else { return }

            state = .purchasing
            do {
                let customerInfo = try await purchaseService.purchase(package: package)
                // Record upgrade on card (best effort - even if this fails, RevenueCat has the purchase)
                await recordCardUpgrade()
                state = .success
            } catch {
                state = .error(error.localizedDescription)
            }
        }

        /// Restore previous purchases
        public func restore() async {
            state = .loading
            do {
                _ = try await purchaseService.restorePurchases()
                // Check if card should be upgraded after restore
                await recordCardUpgrade()
                state = .success
            } catch {
                state = .error(error.localizedDescription)
            }
        }

        private func recordCardUpgrade() async {
            // Update card's maxParticipants to unlimited (999)
            // This is tracked locally - could also use webhook for server-side
            do {
                try await CardService().updateCardMaxParticipants(cardId: cardId, maxParticipants: 999)
            } catch {
                // Best effort - purchase succeeded even if local record fails
                print("Failed to record card upgrade: \(error)")
            }
        }

        var priceString: String? {
            if case .ready(let package) = state {
                return package.localizedPriceString
            }
            return nil
        }
    }
    ```

    /// Updates a card's maximum participant limit (for upgrades).
    public func updateCardMaxParticipants(cardId: UUID, maxParticipants: Int) async throws {
        do {
            try await supabase
                .from("cards")
                .update(["max_participants": maxParticipants])
                .eq("id", value: cardId)
                .execute()

            #if DEBUG
            print("✅ Updated card \(cardId) maxParticipants to: \(maxParticipants)")
            #endif
        } catch {
            throw CardError.updateFailed(error.localizedDescription)
        }
    }

    import SwiftUI
    import RevenueCat

    struct CardUpgradeView: View {
        let card: Card
        let currentParticipantCount: Int
        @State private var viewModel: UpgradeViewModel
        @Environment(\.dismiss) private var dismiss

        init(card: Card, currentParticipantCount: Int) {
            self.card = card
            self.currentParticipantCount = currentParticipantCount
            self._viewModel = State(initialValue: UpgradeViewModel(cardId: card.id))
        }

        var body: some View {
            VStack(spacing: 24) {
                // Header
                VStack(spacing: 8) {
                    Image(systemName: "person.3.fill")
                        .font(.system(size: 48))
                        .foregroundStyle(.toyPrimary)

                    Text("Upgrade Your Card")
                        .font(.custom("DMSerifDisplay-Regular", size: 28))

                    Text("Unlock unlimited participants")
                        .foregroundStyle(.secondary)
                }

                // Current status
                VStack(spacing: 4) {
                    Text("\(currentParticipantCount) of \(card.maxParticipants)")
                        .font(.title2.bold())
                    Text("participants used")
                        .foregroundStyle(.secondary)
                }
                .padding()
                .background(.toySurface)
                .cornerRadius(12)

                // Benefits
                VStack(alignment: .leading, spacing: 12) {
                    benefitRow(icon: "infinity", text: "Unlimited participants")
                    benefitRow(icon: "star.fill", text: "No restrictions on card")
                    benefitRow(icon: "heart.fill", text: "Support TOY development")
                }
                .padding()

                Spacer()

                // Action buttons
                VStack(spacing: 12) {
                    switch viewModel.state {
                    case .idle, .loading:
                        ProgressView()
                            .task { await viewModel.loadOffering() }

                    case .ready(let package):
                        TOYButton("Upgrade - \(package.localizedPriceString)") {
                            Task { await viewModel.purchase() }
                        }
                        .buttonStyle(.primary)

                    case .purchasing:
                        TOYButton("Processing...") {}
                            .disabled(true)

                    case .success:
                        VStack(spacing: 8) {
                            Image(systemName: "checkmark.circle.fill")
                                .font(.system(size: 48))
                                .foregroundStyle(.green)
                            Text("Upgrade Complete!")
                                .font(.headline)
                        }
                        .onAppear {
                            DispatchQueue.main.asyncAfter(deadline: .now() + 1.5) {
                                dismiss()
                            }
                        }

                    case .error(let message):
                        VStack(spacing: 8) {
                            Text(message)
                                .foregroundStyle(.red)
                            TOYButton("Try Again") {
                                Task { await viewModel.loadOffering() }
                            }
                        }
                    }

                    Button("Restore Purchases") {
                        Task { await viewModel.restore() }
                    }
                    .font(.footnote)
                    .foregroundStyle(.secondary)
                }
            }
            .padding()
        }

        private func benefitRow(icon: String, text: String) -> some View {
            HStack(spacing: 12) {
                Image(systemName: icon)
                    .foregroundStyle(.toyPrimary)
                Text(text)
            }
        }
    }
    ```
