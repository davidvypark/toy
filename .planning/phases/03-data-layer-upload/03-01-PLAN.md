Pattern from research:
```swift
actor StorageService {
    func uploadVideo(fileURL: URL, clipId: UUID) async throws -> String {
        let bucket = supabase.storage.from("clips")
        let path = "\(clipId).mov"
        let fileData = try Data(contentsOf: fileURL)

        try await bucket.upload(
            path: path,
            file: fileData,
            options: FileOptions(
                cacheControl: "3600",
                contentType: "video/quicktime",
                upsert: false
            )
        )
        return path
    }
}
```

   - UPDATE policy: authenticated users can update (for upsert if needed later)

SQL from research:
```sql
-- Create clips bucket (private by default)
INSERT INTO storage.buckets (id, name, public)
VALUES ('clips', 'clips', false)
ON CONFLICT (id) DO NOTHING;

-- Allow authenticated users to upload clips
CREATE POLICY "authenticated_insert_clips"
ON storage.objects FOR INSERT
TO authenticated
WITH CHECK (bucket_id = 'clips');

-- Allow authenticated users to read (for signed URL generation)
CREATE POLICY "authenticated_select_clips"
ON storage.objects FOR SELECT
TO authenticated
USING (bucket_id = 'clips');

-- Allow authenticated users to update (for upsert)
CREATE POLICY "authenticated_update_clips"
ON storage.objects FOR UPDATE
TO authenticated
USING (bucket_id = 'clips');

-- Allow authenticated users to delete
CREATE POLICY "authenticated_delete_clips"
ON storage.objects FOR DELETE
TO authenticated
USING (bucket_id = 'clips');
```

IMPORTANT: This migration must be run in Supabase SQL Editor or via `supabase db push`. It cannot be applied automatically from the app.
