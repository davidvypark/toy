Create CameraPreview.swift that wraps AVCaptureVideoPreviewLayer for SwiftUI:

Key requirements from research:
- Use UIViewRepresentable to bridge UIKit preview layer to SwiftUI
- Create custom UIView subclass with AVCaptureVideoPreviewLayer as layer class
- Use .resizeAspectFill for video gravity (fills view, crops edges)
- Accept AVCaptureSession as parameter

Implementation:
```swift
import AVFoundation
import SwiftUI
import UIKit

/// SwiftUI view that displays live camera preview from an AVCaptureSession.
public struct CameraPreview: UIViewRepresentable {
    public let session: AVCaptureSession

    public init(session: AVCaptureSession) {
        self.session = session
    }

    public func makeUIView(context: Context) -> PreviewView {
        let view = PreviewView()
        view.previewLayer.session = session
        view.previewLayer.videoGravity = .resizeAspectFill
        return view
    }

    public func updateUIView(_ uiView: PreviewView, context: Context) {
        // Session is set once in makeUIView, no updates needed
    }
}

/// UIView subclass that uses AVCaptureVideoPreviewLayer as its backing layer.
public final class PreviewView: UIView {
    public override class var layerClass: AnyClass {
        AVCaptureVideoPreviewLayer.self
    }

    public var previewLayer: AVCaptureVideoPreviewLayer {
        layer as! AVCaptureVideoPreviewLayer
    }

    public override init(frame: CGRect) {
        super.init(frame: frame)
        backgroundColor = .black
    }

    @available(*, unavailable)
    required init?(coder: NSCoder) {
        fatalError("init(coder:) has not been implemented")
    }
}

#Preview {
    // Preview shows black rectangle since no real session
    CameraPreview(session: AVCaptureSession())
        .aspectRatio(9/16, contentMode: .fit)
        .clipShape(RoundedRectangle(cornerRadius: 16))
}
```

Key points:
- PreviewView uses layerClass override for efficient layer setup
- Black background for when camera not running
- Public access for use in main app
- No connection updates needed (session set once)
